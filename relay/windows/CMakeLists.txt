cmake_minimum_required(VERSION 3.22)
project(ThreeDSRPWindows VERSION 42.0.3 LANGUAGES C CXX RC)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
add_compile_options("$<$<COMPILE_LANGUAGE:C,CXX>:-ffile-prefix-map=${ROOT}=.>")
set(CORE "${ROOT}/relay/vendor/chiaki-upstream")
set(DEPS "${ROOT}/source/chiaki-ng")
add_compile_definitions(UNICODE _UNICODE _WIN32_WINNT=0x0601 WINVER=0x0601 NOMINMAX WIN32_LEAN_AND_MEAN)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
set(USE_SHARED_MBEDTLS_LIBRARY OFF CACHE BOOL "" FORCE)
add_subdirectory("${ROOT}/relay/vendor/mbedtls" mbedtls EXCLUDE_FROM_ALL)
set(GF_FILES gf.c gf_wgen.c gf_w4.c gf_w8.c gf_w16.c gf_w32.c gf_w64.c gf_w128.c gf_rand.c gf_general.c gf_cpu.c)
list(TRANSFORM GF_FILES PREPEND "${DEPS}/third-party/gf-complete/src/")
add_library(gf_complete STATIC ${GF_FILES})
target_include_directories(gf_complete PUBLIC "${DEPS}/third-party/gf-complete/include")
file(GLOB JER "${DEPS}/third-party/jerasure/src/*.c")
add_library(jerasure STATIC ${JER})
target_include_directories(jerasure PUBLIC "${DEPS}/third-party/jerasure/include")
target_link_libraries(jerasure PUBLIC gf_complete)
add_library(nanopb STATIC "${DEPS}/third-party/nanopb/pb_common.c" "${DEPS}/third-party/nanopb/pb_encode.c" "${DEPS}/third-party/nanopb/pb_decode.c")
target_include_directories(nanopb PUBLIC "${DEPS}/third-party/nanopb")
set(FILES common sock session base64 http ctrl rpcrypt takion senkusha streamconnection ecdh launchspec random gkcrypt audio audioreceiver audiosender videoreceiver frameprocessor packetstats discovery congestioncontrol stoppipe reorderqueue discoveryservice feedback feedbacksender controller takionsendbuffer time fec regist orientation bitstream log thread)
list(TRANSFORM FILES PREPEND "${CORE}/lib/src/")
list(TRANSFORM FILES APPEND ".c")
set(ANDROID_CPP "${ROOT}/relay/android/app/src/main/cpp")
add_library(chiaki STATIC ${FILES} "${ANDROID_CPP}/remote_disabled.c" "${ANDROID_CPP}/generated/takion.pb.c")
target_include_directories(chiaki BEFORE PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/generated")
target_include_directories(chiaki PUBLIC "${CORE}/lib/include" "${ANDROID_CPP}/generated" "${ROOT}/relay/common")
target_compile_definitions(chiaki PUBLIC CHIAKI_LIB_ENABLE_MBEDTLS)
target_link_libraries(chiaki PUBLIC mbedtls mbedx509 mbedcrypto nanopb jerasure ws2_32 bcrypt iphlpapi)
set(ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(ENABLE_STATIC ON CACHE BOOL "" FORCE)
set(WITH_TURBOJPEG OFF CACHE BOOL "" FORCE)
set(WITH_TOOLS OFF CACHE BOOL "" FORCE)
set(WITH_TESTS OFF CACHE BOOL "" FORCE)
set(WITH_SIMD OFF CACHE BOOL "" FORCE)
add_library(jpeg-static STATIC IMPORTED)
set_target_properties(jpeg-static PROPERTIES IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/prefix/lib/libjpeg.a")
set(OPUS_BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(OPUS_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(OPUS_INSTALL_PKG_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
set(OPUS_INSTALL_CMAKE_CONFIG_MODULE OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/opus-1.6.1 opus EXCLUDE_FROM_ALL)
add_executable(3DSRP-Companion WIN32 src/main.cpp src/engine.cpp src/storage.cpp src/usage.cpp app.rc)
target_compile_definitions(3DSRP-Companion PRIVATE RELAY_DESKTOP_VERSION="${PROJECT_VERSION}")
target_include_directories(3DSRP-Companion PRIVATE prefix/include "${ROOT}/relay/vendor/libjpeg-turbo-3.1.3/src" "${CMAKE_CURRENT_SOURCE_DIR}/build/jpeg" "${ANDROID_CPP}" vendor/opus-1.6.1/include)
target_link_libraries(3DSRP-Companion PRIVATE chiaki jpeg-static opus "${CMAKE_CURRENT_SOURCE_DIR}/prefix/lib/libavcodec.a" "${CMAKE_CURRENT_SOURCE_DIR}/prefix/lib/libavutil.a" ws2_32 bcrypt iphlpapi winhttp crypt32 comctl32 shell32 ole32 uuid gdi32 winmm)
target_compile_options(3DSRP-Companion PRIVATE -Wall -Wextra -Wno-misleading-indentation)
if(MINGW)
 target_link_options(3DSRP-Companion PRIVATE -static -static-libgcc -static-libstdc++ -municode -Wl,--dynamicbase,--nxcompat,--high-entropy-va)
endif()
